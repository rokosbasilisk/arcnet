from re_arc.dsl import *
def generate_3bdb4ada() -> any:
    diff_lb = 0
    diff_ub = 1
    cols = interval(0, 10, 1)
    h = unifint(diff_lb, diff_ub, (10, 30))
    print(f"h: {h}")
    w = unifint(diff_lb, diff_ub, (10, 30))
    print(f"w: {w}")
    bgc = choice(cols)
    print(f"bgc: {bgc}")
    remcols = remove(bgc, cols)
    gi = canvas(bgc, (h, w))
    num = unifint(diff_lb, diff_ub, (1, 8))
    print(f"num: {num}")
    indss = asindices(gi)
    maxtrials = 4 * num
    tr = 0
    succ = 0
    while succ < num and tr <= maxtrials:
        if len(remcols) == 0 or len(indss) == 0:
            break
        if choice((True, False)):
            oh = 3
            ow = unifint(diff_lb, diff_ub, (1, max(1, w // 2 - 1))) * 2 + 1
            print(f"oh: {oh}, ow: {ow}")
        else:
            ow = 3
            oh = unifint(diff_lb, diff_ub, (1, max(1, h // 2 - 1))) * 2 + 1
            print(f"ow: {ow}, oh: {oh}")
        subs = totuple(sfilter(indss, lambda ij: ij[0] < h - oh and ij[1] < w - ow))
        if len(subs) == 0:
            tr += 1
            continue
        loci, locj = choice(subs)
        print(f"loci: {loci}, locj: {locj}")
        obj = frozenset({(loci, locj), (loci + oh - 1, locj + ow - 1)})
        bd = backdrop(obj)
        col = choice(remcols)
        print(f"col: {col}")
        if bd.issubset(indss):
            remcols = remove(col, remcols)
            gi = fill(gi, col, bd)
            if oh == 3:
                ln = {(loci + 1, j) for j in range(locj+1, locj+ow, 2)}
            else:
                ln = {(j, locj + 1) for j in range(loci+1, loci+oh, 2)}
            gi = fill(gi, bgc, ln)
            succ += 1
            indss = indss - bd
        tr += 1
    return gi
