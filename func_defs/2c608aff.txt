def generate_2c608aff() -> any:
    diff_lb = 0
    diff_ub = 1

    print(f"loci: {loci}")

    print(f"locj: {locj}")
    bgc = choice(cols)
    print(f"bgc: {bgc}")

    ccol = choice(remcols)
    print(f"ccol: {ccol}")
    remcols = remove(ccol, remcols)
    dcol = choice(remcols)
    print(f"dcol: {dcol}")
    bd = backdrop(frozenset({(loci, locj), (loci + boxh - 1, locj + boxw - 1)}))
    gi = canvas(bgc, (h, w))
    gi = fill(gi, ccol, bd)
    reminds = totuple(asindices(gi) - backdrop(outbox(bd)))

    nnoise = unifint(diff_lb, diff_ub, (0, noiseb))
    print(f"nnoise: {nnoise}")

    print(f"noise: {noise}")
    gi = fill(gi, dcol, noise)
    return gi
